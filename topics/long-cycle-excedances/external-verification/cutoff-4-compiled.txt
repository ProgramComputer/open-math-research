def LongCycles.Allowed : {n : ℕ} → ℕ → Equiv.Perm (Fin n) → Prop :=
fun {n} k σ => ∀ (x : Fin n) (j : ℕ), 0 < j → j < k → (σ ^ j) x ≠ x
def LongCycles.excedances : {n : ℕ} → Equiv.Perm (Fin n) → ℕ :=
fun {n} σ => {x | x < σ x}.card
def LongCycles.enumerator : ℕ → ℕ → Polynomial ℝ :=
fun n k => ∑ σ, if LongCycles.Allowed k σ then Polynomial.X ^ LongCycles.excedances σ else 0
def LongCycleFinite.StrictNegativeInterlacing : Polynomial ℝ → Polynomial ℝ → ℕ → Prop :=
fun p q d =>
  ∃ r s,
    p.leadingCoeff ≠ 0 ∧
      q.leadingCoeff ≠ 0 ∧
        p = Polynomial.C p.leadingCoeff * ∏ i, (Polynomial.X - Polynomial.C (r i)) ∧
          q = Polynomial.C q.leadingCoeff * ∏ i, (Polynomial.X - Polynomial.C (s i)) ∧
            StrictMono r ∧
              StrictMono s ∧
                (∀ (i : Fin d), r i < 0) ∧
                  (∀ (i : Fin (d + 1)), s i < 0) ∧ ∀ (i : Fin d), s i.castSucc < r i ∧ r i < s i.succ
def LongCycles.WeakRealInterlacing : Polynomial ℝ → Polynomial ℝ → Prop :=
fun p q =>
  ∃ d r s a b,
    a ≠ 0 ∧
      b ≠ 0 ∧
        p = Polynomial.C a * LongCycles.linearFactors r ∧
          q = Polynomial.C b * LongCycles.linearFactors s ∧
            Monotone r ∧ Monotone s ∧ ∀ (i : Fin d), s i.castSucc ≤ r i ∧ r i ≤ s i.succ
